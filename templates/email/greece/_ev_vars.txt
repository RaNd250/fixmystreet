[%# Shared by submit-ev.txt and submit-ev.html: the complaint to the police about parking on an
    EV-charging bay or that blocks the way (category «Στάθμευση σε φορτιστή EV ή εμπόδιο στη
    διέλευση», email_template "ev"), built from the category's questions; also the same complaint to a
    municipality about a shop's obstacles or a P-72 sign (email_template "katalipsi"). ~%]
[%
paravasi = report.get_extra_field_value('paravasi');
pinakida = report.greece_plate;

ev = paravasi == 'fortistis';
bus = paravasi == 'stasi';
passage = paravasi == 'eisodos' || paravasi == 'diplo';
paravasi_texts = {
    fortistis => 'Στάθμευση οχήματος σε θέση φόρτισης ηλεκτρικών οχημάτων χωρίς να φορτίζει',
    eisodos => 'Στάθμευση οχήματος μπροστά σε είσοδο ή έξοδο οχημάτων (γκαράζ), που εμποδίζει τη διέλευση',
    diplo => 'Διπλή στάθμευση οχήματος ή στάθμευση που κλείνει τον δρόμο και εμποδίζει τη διέλευση',
    stasi => 'Στάθμευση οχήματος σε στάση λεωφορείου, που εμποδίζει τα λεωφορεία και τους επιβάτες',
    empodia => 'Τοποθέτηση παράνομων εμποδίων ή κώνων από κατάστημα',
    p72 => 'Ύπαρξη αυθαίρετης ή μη εγκεκριμένης διαγράμμισης και πινακίδας P-72',
};
paravasi_text = paravasi ? paravasi_texts.$paravasi : '';

sp = report.greece_street_parts(closest_address);
street = sp.street;
number = sp.number;
place = '';
IF closest_address;
    addr = closest_address.address;
    place = addr.suburb OR addr.neighbourhood OR addr.city OR addr.town OR addr.village OR '';
END;
area = [];
area.push(place) IF place;
area.push(dimos_name) IF dimos_name;

subject_place = street ? 'επί της οδού ' _ street : 'στην περιοχή ' _ (area.join(', ') OR 'που φαίνεται στον χάρτη');
IF ev;
    subject = 'Καταγγελία για Παράνομη Κατάληψη Θέσης Φόρτισης Ηλεκτρικών Οχημάτων ' _ subject_place;
    purpose = 'τη φόρτιση ηλεκτρικών οχημάτων';
ELSIF bus;
    subject = 'Καταγγελία για Παράνομη Στάθμευση σε Στάση Λεωφορείου ' _ subject_place;
    purpose = 'τη στάση των λεωφορείων';
ELSIF passage;
    subject = 'Καταγγελία για Παράνομη Στάθμευση που Εμποδίζει τη Διέλευση ' _ subject_place;
    purpose = 'τη διέλευση οχημάτων';
ELSE;
    subject = 'Καταγγελία για Παράνομα Εμπόδια / Σήμανση Στάθμευσης ' _ subject_place;
    purpose = 'την ελεύθερη στάθμευση και διέλευση';
END;

location = street ? 'Επί της οδού ' _ street _ (number ? ' ' _ number : '') : 'Στο σημείο του χάρτη';
location = location _ (area.size ? ', στην περιοχή ' _ area.join(', ') : '');

# The police get the photos with the plates readable (PhotoStorage::Greece); a municipality, when the
# report went to it instead (email_template "katalipsi", Greece.pm greece_route_to_dimos), the blurred ones.
photos = report.greece_photos(email_template == 'katalipsi' ? '' : 'police');
~%]
